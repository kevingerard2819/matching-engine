cmake_minimum_required(VERSION 3.16)
project(matching_engine LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(ENABLE_TSAN "Enable thread sanitizer (separate build from ASan)" OFF)
if(ENABLE_SANITIZERS AND ENABLE_TSAN)
  message(FATAL_ERROR "Use separate builds for address and thread sanitizers")
endif()
find_package(Threads REQUIRED)

add_library(matching_core src/order_book.cpp src/matching_engine.cpp src/order_io.cpp src/feed_protocol.cpp)
target_include_directories(matching_core PUBLIC src)
target_link_libraries(matching_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(matching_core PUBLIC /W4 /permissive-)
else()
  target_compile_options(matching_core PUBLIC -Wall -Wextra -Wpedantic)
  if(ENABLE_SANITIZERS)
    target_compile_options(matching_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(matching_core PUBLIC -fsanitize=address,undefined)
  endif()
  if(ENABLE_TSAN)
    target_compile_options(matching_core PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(matching_core PUBLIC -fsanitize=thread)
  endif()
endif()

add_executable(replay src/replay.cpp)
target_link_libraries(replay PRIVATE matching_core)
add_executable(benchmark bench/benchmark.cpp)
target_link_libraries(benchmark PRIVATE matching_core)
add_executable(generate_orders bench/generate_orders.cpp)
target_link_libraries(generate_orders PRIVATE matching_core)

enable_testing()
add_executable(order_book_tests tests/order_book_tests.cpp)
target_link_libraries(order_book_tests PRIVATE matching_core)
add_test(NAME order_book_tests COMMAND order_book_tests)
add_test(NAME replay_example COMMAND replay ${CMAKE_CURRENT_SOURCE_DIR}/data/example.orders)
set_tests_properties(replay_example PROPERTIES PASS_REGULAR_EXPRESSION "trades=1 resting_orders=1")

add_executable(feed_tests tests/feed_tests.cpp)
target_link_libraries(feed_tests PRIVATE matching_core)
add_test(NAME feed_tests COMMAND feed_tests)
set_tests_properties(feed_tests PROPERTIES TIMEOUT 30)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME benchmark_equivalence COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/benchmark_equivalence.py
    $<TARGET_FILE:benchmark> $<TARGET_FILE:generate_orders>)
  set_tests_properties(benchmark_equivalence PROPERTIES TIMEOUT 60)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(udp_feed src/udp_feed.cpp)
  target_link_libraries(udp_feed PRIVATE matching_core)
  add_executable(udp_sender src/udp_sender.cpp)
  target_link_libraries(udp_sender PRIVATE matching_core)
  if(Python3_Interpreter_FOUND)
    add_test(NAME udp_integration COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/udp_integration.py
      $<TARGET_FILE:udp_feed> $<TARGET_FILE:udp_sender> $<TARGET_FILE:replay>)
    set_tests_properties(udp_integration PROPERTIES TIMEOUT 45)
  endif()
endif()
